1. Select External Volumes#

1.1 Select Volumes by Defining a Point#

Select the foundation by defining a point on foundation.

point = Point(18, 15, 0)

selectByPoint = model.Selection.getExternalVolumes(point)
selectByPoint[0].getName()

'Foundation'

1.2 Select Volumes by Definiing a Polyline#

Select tunnel, foundation and soil by create a polyline going through all three volumes.

polyline = Polyline([Point(10, 2, -10), Point(18, 15, 0)])

selectByPolyline = model.Selection.getExternalVolumes(polyline)
for volume in selectByPolyline:
    print(volume.getName())

Underground Tunnel
Soil
Foundation

1.3 Select Volumes by Defining a Cube#

Define a cube which is slightly larger than the tunnel to include some tolerance.

cube = Cube(Point(8.5, 0, -11.5), Point(11.5, 30, -8.5))

To get external volumes within a defined volume, include intersecting can be defined. If include intersecting is True, volumes that attach the defining volume will be selected. Otherwise, intersecting volumes will not be included.

If intersecting volumes are included, both underground tunnel and soil volumes will be chosen.

selectByCube = model.Selection.getExternalVolumes(cube)
for volume in selectByCube:
    print(volume.getName())

Underground Tunnel
Soil

If intersecting volumes are not included, only underground tunnel will be selected.

selectByCubeWithoutIntersection = model.Selection.getExternalVolumes(cube, False)
for volume in selectByCubeWithoutIntersection:
    print(volume.getName())

# External Volume and Selection Script Examples
# Download the [blankModel.rs3v3](https://github.com/Rocscience/rs3-scripting/tree/main/docs/example_code/example_models/blankModel.rs3v3) for this example.
from rs3.RS3Modeler import RS3Modeler
from rs3.properties.PropertyEnums import *
from rs3.Geometry import Point, Polyline, Cube, Cylinder, Sphere
from rs3.CommonEnums import *
import os
import shutil
# Get the current folder directory.
current_dir = os.path.dirname(os.path.abspath(""))
# Specify a port number that is not in use and start the RS3 program.
port = 60104
RS3Modeler.startApplication(port)
# Connect with the RS3 Modeler.
modeler = RS3Modeler(port)
# For the demonstration purposes only, the model is copied for reuse. 
# You may change the model path to your own model path.
blankModelPath = rf"{current_dir}\example_models\blankModel.rs3v3"
copiedModelPath = rf"{current_dir}\example_models\copiedModel.rs3v3"
_ = shutil.copy(blankModelPath, copiedModelPath)
# Open the copied model file.
model = modeler.openFile(copiedModelPath)

# 1. Select External Volumes

# 1.1 Select Volumes by Defining a Point
# Select the foundation by defining a point on foundation.
point = Point(18, 15, 0)
selectByPoint = model.Selection.getExternalVolumes(point)

# 1.2 Select Volumes by Definiing a Polyline
# Select tunnel, foundation and soil by create a polyline going through all three volumes.
polyline = Polyline([Point(10, 2, -10), Point(18, 15, 0)])
selectByPolyline = model.Selection.getExternalVolumes(polyline)
for volume in selectByPolyline:
    print(volume.getName())
    
# 1.3 Select Volumes by Defining a Cube
# Define a cube which is slightly larger than the tunnel to include some tolerance.
cube = Cube(Point(8.5, 0, -11.5), Point(11.5, 30, -8.5))
# To get external volumes within a defined volume, include intersecting can be defined. 
# If include intersecting is True, volumes that attach the defining volume will be selected. 
# Otherwise, intersecting volumes will not be included.
# If intersecting volumes are included, both underground tunnel and soil volumes will be chosen.
selectByCube = model.Selection.getExternalVolumes(cube)
for volume in selectByCube:
    print(volume.getName())
# If intersecting volumes are not included, only underground tunnel will be selected.
selectByCubeWithoutIntersection = model.Selection.getExternalVolumes(cube, False)
for volume in selectByCubeWithoutIntersection:
    print(volume.getName())
    
# 1.4 Select Volumes by Defining a Cylinder
# Create a cylinder which is slightly larger than the actual tunnel
cylinder = Cylinder(Point(10, 0, -10), Point(10, 30, -10), 1.5)
selectByCylinder = model.Selection.getExternalVolumes(cylinder)
for volume in selectByCylinder:
    print(volume.getName())
selectByCylinderWithoutIntersection = model.Selection.getExternalVolumes(
    cylinder, False
)
for volume in selectByCylinderWithoutIntersection:
    print(volume.getName())
    
# 1.5 Select Volumes by Defining a Sphere
sphere = Sphere(Point(17.5, 15, 0), 5.5)
# Create a sphere that is slightly larger than the foundation and include intersecting volumes. 
# Both foundation and soil will be selected.
selectBySphere = model.Selection.getExternalVolumes(sphere)
for volume in selectBySphere:
    print(volume.getName())
# If exclude intersecting volumes, foundation will be the only one that's fully inside the defined region.
selectBySphereWithoutIntersection = model.Selection.getExternalVolumes(sphere, False)
for volume in selectBySphereWithoutIntersection:
    print(volume.getName())
    
# 2. Change External Volume Properties
# Once external volume is selected, external volume properties can be changed. 
# Let's first select the soil volume.
soil = model.Selection.getExternalVolumes(Point(22, 0, -12))[0]
# Change the external volume name to "Material 3 Soil".
soil.setName("Material 4 Soil")
# Get material 4.
material4 = model.getAllMaterialProperties()[3]
material4Name = material4.getMaterialName()
# Set the soil model to Material 4 since stage 1.
soil.setAppliedMaterialProperty(1, material4Name)
# Set the soil role to CONSTRUCTION.
soil.setRole(ExternalVolumeRoles.CONSTRUCTION)
# Change the soil role back to GEOLOGY.
soil.setRole(ExternalVolumeRoles.GEOLOGY)

# 3. Attached Volumes
# This function is for selecting volumes attached to the given volumes for easier selection.
# You may provide an external volume object to look up attached volumes. For example, tunnel 
# and foundation are attached to the soil.
attachedVolumesbyExternalVolume = model.Selection.getAttachedExternalVolumes(soil)
for volume in attachedVolumesbyExternalVolume:
    print(volume.getName())
# Without getting the external volume first, you may select the external volume by name as well.
attachedVolumesbyExternalVolumeName = (
    model.Selection.getAttachedExternalVolumesByVolumeName(soil.getName())
)
for volume in attachedVolumesbyExternalVolumeName:
    print(volume.getName())
    
# 4. Volumes and Areas
# Get the volume and surface area of the tunnel.
tunnel = model.getExternalVolumeByName("Underground Tunnel")
print(f"The volume of the tunnel is {round(tunnel.getVolume(), 2)} m^3.")
print(
    f"The surface area of the tunnel is {round(tunnel.getTotalSurfaceArea(), 2)} m^2."
)

# Save and close the model.
model.close(False)
